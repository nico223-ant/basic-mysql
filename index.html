Panduan Dasar MySQL

Langkah-langkah instalasi dan perintah dasar pengelolaan database

1
Install MySQL Server dan Client
apt-get install mysql-server mysql-client
2
Masuk ke MySQL
mysql -u root -p
3
Melihat Database
show databases;
4
Membuat Database Bernama "sekolah"
create database sekolah;
5
Melihat Database (Memastikan terbuat)
show databases;
6
Masuk ke Database "sekolah"
use sekolah;
7
Membuat Tabel Bernama "siswa"
create table siswa (nis varchar(9), nama varchar(50), kelas varchar(15), alamat varchar(100));
8
Melihat Tabel
show tables;
9
Melihat Struktur Tabel
desc siswa;
10
Mengisi Tabel (Insert Data)
insert into siswa values ('242510753','Andi','XI TKJ','Bandung');
11
Melihat Isi Tabel
select * from siswa;
